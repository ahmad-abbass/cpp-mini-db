cmake_minimum_required(VERSION 3.20)
project(cpp_mini_db VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)
option(MINIDB_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(MINIDB_BUILD_BENCHMARKS "Build the verified embedded benchmark" ON)
option(MINIDB_BUILD_FUZZER "Build the Clang libFuzzer WAL decoder harness" OFF)

function(minidb_configure_target target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()

    if(MINIDB_ENABLE_SANITIZERS)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            message(FATAL_ERROR "Sanitizers require GCC or Clang for this project")
        endif()
        target_compile_options(${target} PRIVATE
            -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

add_library(minidb STATIC src/key_value_store.cpp)
add_library(minidb::minidb ALIAS minidb)
target_include_directories(minidb PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(minidb PUBLIC cxx_std_17)
minidb_configure_target(minidb)

if(UNIX)
    find_package(Threads REQUIRED)
    target_sources(minidb PRIVATE src/database.cpp src/wal.cpp src/command.cpp)
    target_link_libraries(minidb PUBLIC Threads::Threads)
    if(MINIDB_BUILD_FUZZER)
        if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            message(FATAL_ERROR "MINIDB_BUILD_FUZZER requires Clang")
        endif()
        add_executable(wal_fuzzer fuzz/wal_decode_fuzzer.cpp src/wal.cpp src/database.cpp)
        target_include_directories(wal_fuzzer PRIVATE include src)
        target_compile_features(wal_fuzzer PRIVATE cxx_std_17)
        target_link_libraries(wal_fuzzer PRIVATE Threads::Threads)
        minidb_configure_target(wal_fuzzer)
        target_compile_options(wal_fuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
        target_link_options(wal_fuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
    endif()
    add_executable(minidb_cli apps/cli.cpp)
    target_link_libraries(minidb_cli PRIVATE minidb::minidb)
    minidb_configure_target(minidb_cli)
    if(MINIDB_BUILD_BENCHMARKS)
        add_executable(minidb_bench benchmarks/database_benchmark.cpp)
        target_link_libraries(minidb_bench PRIVATE minidb::minidb)
        minidb_configure_target(minidb_bench)
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(minidb_server apps/server.cpp)
        target_link_libraries(minidb_server PRIVATE minidb::minidb)
        minidb_configure_target(minidb_server)
    endif()
endif()

add_executable(minidb_example examples/basic_usage.cpp)
target_link_libraries(minidb_example PRIVATE minidb::minidb)
minidb_configure_target(minidb_example)

    target_link_libraries(minidb_tests PRIVATE minidb::minidb)
    minidb_configure_target(minidb_tests)
    add_test(NAME key_value_store COMMAND minidb_tests)
    add_test(NAME basic_usage COMMAND minidb_example)
    if(UNIX)

        target_include_directories(database_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")

            "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py" $<TARGET_FILE:minidb_cli>)

        add_test(NAME wal_inspector COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/wal_inspector_tests.py" $<TARGET_FILE:minidb_cli>)

                "${CMAKE_CURRENT_SOURCE_DIR}/tests/server_tests.py" $<TARGET_FILE:minidb_server>)
            set_tests_properties(tcp_server PROPERTIES TIMEOUT 120)
            add_test(NAME complete_demo COMMAND ${Python3_EXECUTABLE}
                "${CMAKE_CURRENT_SOURCE_DIR}/tools/demo.py" --server $<TARGET_FILE:minidb_server>)

                add_test(NAME embedded_benchmark_smoke COMMAND minidb_bench --operations 20)

                set_tests_properties(embedded_benchmark_smoke network_benchmark_smoke PROPERTIES TIMEOUT 30)
            endif()
        endif()
    endif()
